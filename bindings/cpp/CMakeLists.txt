cmake_minimum_required(VERSION 3.20)
project(Tapirscan VERSION 1.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(BARCODE_MODE medium CACHE STRING "Scanner mode: low, medium, high or very-high")
set_property(CACHE BARCODE_MODE PROPERTY STRINGS low medium high very-high)
if(NOT BARCODE_MODE MATCHES "^(low|medium|high|very-high)$")
  message(FATAL_ERROR "BARCODE_MODE must be low, medium, high or very-high")
endif()
set(BARCODE_NATIVE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../build/native" CACHE PATH "Directory containing native libraries")
string(REPLACE "-" "_" BARCODE_LIB_MODE "${BARCODE_MODE}")
set(BARCODE_NATIVE_NAME "${CMAKE_SHARED_LIBRARY_PREFIX}tapirscan_${BARCODE_LIB_MODE}${CMAKE_SHARED_LIBRARY_SUFFIX}")
if(NOT EXISTS "${BARCODE_NATIVE_DIR}/${BARCODE_NATIVE_NAME}")
  message(FATAL_ERROR "Build native libraries first: python3 scripts/build_native.py ${BARCODE_MODE}")
endif()
add_library(tapirscan::native SHARED IMPORTED GLOBAL)
set_target_properties(tapirscan::native PROPERTIES IMPORTED_LOCATION "${BARCODE_NATIVE_DIR}/${BARCODE_NATIVE_NAME}")
set(BARCODE_NATIVE_SONAME "${BARCODE_NATIVE_NAME}")
if(APPLE)
  set(BARCODE_NATIVE_SONAME "@rpath/${BARCODE_NATIVE_NAME}")
endif()
if(UNIX)
  set_target_properties(tapirscan::native PROPERTIES IMPORTED_SONAME "${BARCODE_NATIVE_SONAME}")
endif()
if(WIN32)
  set_target_properties(tapirscan::native PROPERTIES IMPORTED_IMPLIB "${BARCODE_NATIVE_DIR}/tapirscan_${BARCODE_LIB_MODE}.dll.lib")
endif()
add_library(tapirscan_cpp INTERFACE)
set_target_properties(tapirscan_cpp PROPERTIES EXPORT_NAME cpp)
add_library(tapirscan::cpp ALIAS tapirscan_cpp)
target_compile_features(tapirscan_cpp INTERFACE cxx_std_17)
target_include_directories(tapirscan_cpp INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}/include>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}/../c/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(tapirscan_cpp INTERFACE tapirscan::native)
option(BARCODE_BUILD_TESTS "Build smoke test and scan example" ON)
if(BARCODE_BUILD_TESTS)
  enable_testing()
  add_executable(scan_raw examples/scan_raw.cpp)
  target_link_libraries(scan_raw PRIVATE tapirscan::cpp)
  add_executable(c_smoke ../c/tests/smoke.c)
  target_include_directories(c_smoke PRIVATE ../c/include)
  target_link_libraries(c_smoke PRIVATE tapirscan::native)
  # Assert-based tests must execute even in Release builds.
  if(MSVC)
    target_compile_options(c_smoke PRIVATE /UNDEBUG)
    target_compile_options(scan_raw PRIVATE /UNDEBUG)
  else()
    target_compile_options(c_smoke PRIVATE -UNDEBUG)
    target_compile_options(scan_raw PRIVATE -UNDEBUG)
  endif()
  add_test(NAME c_lifetime_and_buffers COMMAND c_smoke)
endif()
install(TARGETS tapirscan_cpp EXPORT TapirscanTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES ../c/include/tapirscan.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES "${BARCODE_NATIVE_DIR}/${BARCODE_NATIVE_NAME}" DESTINATION ${CMAKE_INSTALL_LIBDIR})
if(WIN32)
  install(FILES "${BARCODE_NATIVE_DIR}/tapirscan_${BARCODE_LIB_MODE}.dll.lib" DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
configure_package_config_file(TapirscanConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/TapirscanConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Tapirscan)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/TapirscanConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT TapirscanTargets NAMESPACE tapirscan:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Tapirscan)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/TapirscanConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/TapirscanConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Tapirscan)

install(FILES ../../LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/tapirscan)
