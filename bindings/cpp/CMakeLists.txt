cmake_minimum_required(VERSION 3.20)
project(Tapirscan VERSION 1.3.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(TAPIRSCAN_NATIVE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../build/native" CACHE PATH "Directory containing the native library")
set(TAPIRSCAN_NATIVE_NAME "${CMAKE_SHARED_LIBRARY_PREFIX}tapirscan${CMAKE_SHARED_LIBRARY_SUFFIX}")
if(NOT EXISTS "${TAPIRSCAN_NATIVE_DIR}/${TAPIRSCAN_NATIVE_NAME}")
  message(FATAL_ERROR "Build the native library first: python3 scripts/build_native.py")
endif()
add_library(tapirscan::native SHARED IMPORTED GLOBAL)
set_target_properties(tapirscan::native PROPERTIES
  IMPORTED_LOCATION "${TAPIRSCAN_NATIVE_DIR}/${TAPIRSCAN_NATIVE_NAME}"
  INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_LIST_DIR}/../c/include")
set(TAPIRSCAN_NATIVE_SONAME "${TAPIRSCAN_NATIVE_NAME}")
if(APPLE)
  set(TAPIRSCAN_NATIVE_SONAME "@rpath/${TAPIRSCAN_NATIVE_NAME}")
endif()
if(UNIX)
  set_target_properties(tapirscan::native PROPERTIES IMPORTED_SONAME "${TAPIRSCAN_NATIVE_SONAME}")
endif()
if(WIN32)
  set_target_properties(tapirscan::native PROPERTIES IMPORTED_IMPLIB "${TAPIRSCAN_NATIVE_DIR}/tapirscan.dll.lib")
endif()
add_library(tapirscan_cpp INTERFACE)
set_target_properties(tapirscan_cpp PROPERTIES EXPORT_NAME cpp)
add_library(tapirscan::cpp ALIAS tapirscan_cpp)
target_compile_features(tapirscan_cpp INTERFACE cxx_std_17)
target_include_directories(tapirscan_cpp INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}/include>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}/../c/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(tapirscan_cpp INTERFACE tapirscan::native)
option(TAPIRSCAN_BUILD_TESTS "Build the C smoke test and the C++ scan example" ON)
if(TAPIRSCAN_BUILD_TESTS)
  enable_testing()
  add_executable(scan_raw examples/scan_raw.cpp)
  target_link_libraries(scan_raw PRIVATE tapirscan::cpp)
  add_executable(cpp_api tests/api.cpp)
  target_link_libraries(cpp_api PRIVATE tapirscan::cpp)
  add_test(NAME cpp_api_contract COMMAND cpp_api)
  add_executable(c_smoke ../c/tests/smoke.c)
  target_link_libraries(c_smoke PRIVATE tapirscan::native)
  # Assert-based tests must execute even in Release builds.
  if(MSVC)
    target_compile_options(c_smoke PRIVATE /UNDEBUG)
    target_compile_options(cpp_api PRIVATE /UNDEBUG)
  else()
    target_compile_options(c_smoke PRIVATE -UNDEBUG)
    target_compile_options(cpp_api PRIVATE -UNDEBUG)
  endif()
  add_test(NAME c_lifetime_and_buffers COMMAND c_smoke)
  # Windows finds DLLs next to the executable; copy the native library there.
  if(WIN32)
    foreach(target scan_raw cpp_api c_smoke)
      add_custom_command(TARGET ${target} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
          "${TAPIRSCAN_NATIVE_DIR}/${TAPIRSCAN_NATIVE_NAME}" "$<TARGET_FILE_DIR:${target}>")
    endforeach()
  endif()
endif()
install(TARGETS tapirscan_cpp EXPORT TapirscanTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES ../c/include/tapirscan.h ../c/include/tapirscan_formats.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES "${TAPIRSCAN_NATIVE_DIR}/${TAPIRSCAN_NATIVE_NAME}" DESTINATION ${CMAKE_INSTALL_LIBDIR})
if(WIN32)
  install(FILES "${TAPIRSCAN_NATIVE_DIR}/tapirscan.dll.lib" DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
# PATH_VARS resolves relative and absolute install directories correctly.
configure_package_config_file(TapirscanConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/TapirscanConfig.cmake"
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Tapirscan
  PATH_VARS CMAKE_INSTALL_LIBDIR CMAKE_INSTALL_INCLUDEDIR)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/TapirscanConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT TapirscanTargets NAMESPACE tapirscan:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Tapirscan)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/TapirscanConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/TapirscanConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Tapirscan)

install(FILES ../../LICENSE ../../multiformat/THIRD_PARTY_NOTICES.md
  DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/tapirscan)
